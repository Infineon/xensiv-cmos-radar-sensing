Use-Case API

Example

The following pseudo-code example shows how to run the use-case library. Please refer to the full embedded example how to receive radar frames on an embedded platform.

/* Get default configuration. Optionally modify it before initializing. */
ifx_usecase_config config;
ifx_usecase_get_default_configuration(&config);

/* Optionally adjust user settings (e.g. mounting height) before init:  */
/* ifx_usecase_settings settings;                                        */
/* ifx_usecase_get_default_settings(&settings);                          */
/* settings.sensor_mounting.mounting_height_m = 2.5f;                   */
/* ifx_usecase_settings_set_settings(&config, &settings);                */

/* Init use-case algorithm */
ifx_usecase_handle usecase_handle = ifx_usecase_init(&config);

bool reset = false; /* set to true to reset the pipeline state on the next frame */

for(;;)
{
    /* wait for new radar frame */
    const uint16_t* radar_frame_data = ...;
    uint32_t timestamp_ms = ...;

    /* run usecase with radar frame */
    ifx_event_flags event_flags = IFX_EVENT_FLAG_NONE;
    ifx_usecase_process(usecase_handle, reset, radar_frame_data, &event_flags);
    reset = false;

    /* check for presence events */
    const ifx_presence_event* presence_event = NULL;
    if(event_flags & IFX_EVENT_FLAG_PRESENCE)
    {
        if(ifx_usecase_get_presence_event(usecase_handle, &presence_event) >= 0)
        {
            /* handle presence event */
        }
    }

    /* check for tracking events */
    const ifx_tracking_event* tracking_events = NULL;
    uint32_t tracking_events_count = 0;
    if(event_flags & IFX_EVENT_FLAG_TRACKING)
    {
        if(ifx_usecase_get_tracking_events(usecase_handle, &tracking_events, &tracking_events_count) >= 0)
        {
            /* handle tracking events */
        }
    }

    /* check for detection events */
    const ifx_detection* detections = NULL;
    uint32_t detections_count = 0;
    if(event_flags & IFX_EVENT_FLAG_DETECTIONS)
    {
        if(ifx_usecase_get_detections(usecase_handle, &detections, &detections_count) >= 0)
        {
            /* handle detection events */
        }
    }

    /* serialize all raised events as a JSONL line into a buffer */
    char json_buffer[1024];
    ifx_print_event(json_buffer, sizeof(json_buffer), usecase_handle, event_flags, timestamp_ms);
}

/* de-initialize the library */
ifx_usecase_deinit(&usecase_handle);

int ifx_usecase_get_detections(ifx_usecase_handle handle, const ifx_detection **detections, uint32_t *detections_count)

Gets a pointer to a list of detection events.

See also

ifx_detection

Note

The event is only valid until the next call to ifx_usecase_process.

Parameters:

  • handle – Usecase handle

  • detections – [out] A pointer to a pointer of type ifx_detection, which will point to the internal retrieved detection event structure.

  • detections_count – [out] Number of detections returned through detections pointer.

Returns:

0 on success, a negative value on error.

int ifx_print_event(char *json_buffer, size_t json_buffer_size, ifx_usecase_handle handle, ifx_event_flags events_flags, uint32_t timestamp)

Print events as JSONL format to a buffer.

Parameters:

  • json_buffer – Output buffer

  • json_buffer_size – Output buffer size

  • handle – Usecase handle

  • events_flags – Event flags that shall be written to the output buffer

  • timestamp – Timestamp in milliseconds

  • elapsed_time – Elapsed time for the use-case processing in milliseconds

  • frames_dropped – Dropped frames since last use-case processing

Returns:

IFX_USECASE_API

Note

IFX_EVENT_FLAG_ZONING is defined in ifx_event_flags but is reserved for future use. No corresponding getter exists in the current API; do not test for this flag.
